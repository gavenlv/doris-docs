#!/bin/bash

# ==========================================
# Doris 安全镜像 - 本地验证指南
# ==========================================
# 由于 Docker 环境问题，请在本地手动执行以下步骤

echo "=========================================="
echo " Doris 安全镜像 - 本地验证指南"
echo "=========================================="
echo ""

echo "步骤 1: 启动 Nexus"
echo "-------------------"
echo "运行以下命令启动 Nexus:"
echo ""
echo "  docker run -d --name doris-nexus \\"
echo "    -p 8081:8081 \\"
echo "    -p 8082:8082 \\"
echo "    sonatype/nexus3:latest"
echo ""
echo "等待 1-2 分钟让 Nexus 启动完成"
echo ""

echo "步骤 2: 获取 Nexus 初始密码"
echo "-------------------"
echo "  docker exec doris-nexus cat /nexus-data/admin.password"
echo ""

echo "步骤 3: 配置 Nexus"
echo "-------------------"
echo "1. 访问 http://localhost:8081"
echo "2. 使用 admin/初始密码 登录"
echo "3. 创建 Docker Hosted Repository:"
echo "   - Name: doris-docker"
echo "   - HTTP Port: 8082"
echo "4. 创建 Raw Hosted Repository:"
echo "   - Name: doris-packages"
echo ""

echo "步骤 4: 下载离线包"
echo "-------------------"
echo "  mkdir -p offline-packages/doris-fe offline-packages/doris-be offline-packages/foundationdb"
echo ""
echo "  # 下载 Doris FE"
echo "  wget -q https://archive.apache.org/dist/doris/3.1.4/apache-doris-fe-3.1.4-bin.tar.gz \\"
echo "    -O offline-packages/doris-fe/"
echo ""
echo "  # 下载 Doris BE"
echo "  wget -q https://archive.apache.org/dist/doris/3.1.4/apache-doris-be-3.1.4-bin-x86_64.tar.gz \\"
echo "    -O offline-packages/doris-be/"
echo ""
echo "  # 下载 FoundationDB"
echo "  wget -q https://github.com/apple/foundationdb/releases/download/7.1.37/foundationdb-clients_7.1.37-1_amd64.deb \\"
echo "    -O offline-packages/foundationdb/"
echo "  wget -q https://github.com/apple/foundationdb/releases/download/7.1.37/foundationdb-server_7.1.37-1_amd64.deb \\"
echo "    -O offline-packages/foundationdb/"
echo ""

echo "步骤 5: 构建镜像"
echo "-------------------"
echo "  # 构建 FE"
echo "  cd doris-gke-cluster-build-from-ubuntu-latest"
echo "  docker build \\"
echo "    --build-arg DORIS_VERSION=3.1.4 \\"
echo "    --build-arg OFFLINE_PATH=/offline-packages \\"
echo "    -f docker/fe/Dockerfile \\"
echo "    -t doris-fe:3.1.4 \\"
echo "    ."
echo ""
echo "  # 构建 BE"
echo "  docker build \\"
echo "    --build-arg DORIS_VERSION=3.1.4 \\"
echo "    --build-arg OFFLINE_PATH=/offline-packages \\"
echo "    -f docker/be/Dockerfile \\"
echo "    -t doris-be:3.1.4 \\"
echo "    ."
echo ""

echo "步骤 6: 推送到 Nexus"
echo "-------------------"
echo "  # 登录 Nexus"
echo "  docker login localhost:8082 -u admin -p adminadmin"
echo ""
echo "  # Tag 和推送"
echo "  docker tag doris-fe:3.1.4 localhost:8082/doris/fe:3.1.4-secure"
echo "  docker tag doris-be:3.1.4 localhost:8082/doris/be:3.1.4-secure"
echo ""
echo "  docker push localhost:8082/doris/fe:3.1.4-secure"
echo "  docker push localhost:8082/doris/be:3.1.4-secure"
echo ""

echo "步骤 7: 验证"
echo "-------------------"
echo "  # 从 Nexus 拉取"
echo "  docker pull localhost:8082/doris/fe:3.1.4-secure"
echo "  docker pull localhost:8082/doris/be:3.1.4-secure"
echo ""
echo "  # 检查镜像"
echo "  docker images | grep doris"
echo ""

echo "=========================================="
echo " 验证完成!"
echo "=========================================="
